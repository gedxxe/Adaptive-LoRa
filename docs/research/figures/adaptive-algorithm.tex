\begin{algorithm}{1}{Gateway scheduling}
  \Step{$\Expire_\iota(\phi)$ for each roster position $\iota$}
  \Step{$\Tx{\ctrl}{\mathrm{Beacon}(\phi,\Omega)}$}
  \Step{$(\Theta_0,\boldsymbol{\phi}_{\mathrm{ack}}) \gets \Requests(\phi,\Omega)$}
  \Step{\textbf{if} $\Theta_0=\varnothing$ \textbf{then return}}
  \Step{$(\boldsymbol{\sigma}^{\mathrm{want}},\boldsymbol{\sigma}^{\mathrm{res}})
    \gets \Recover(\phi,\Omega,\Theta_0,\boldsymbol{\phi}_{\mathrm{ack}})$}
  \Step{$\boldsymbol{\sigma}^{\mathrm{res}}\gets
    \Reserve(\Omega,\boldsymbol{\sigma}^{\mathrm{want}},\boldsymbol{\sigma}^{\mathrm{res}})$}
  \Step{\hspace{1em}subject to $\tau_{\mathrm{used}}(\Omega,\boldsymbol{\sigma}^{\mathrm{res}})\le\tau_{\mathrm{frame}}$}
  \Step{$\sigma_\iota\gets\min(\sigma_\iota^{\mathrm{want}},\sigma_\iota^{\mathrm{res}}),\quad \iota\in\Theta_0$}
  \Step{$\Tx{\ctrl}{\mathrm{Grant}(\phi,\Theta_0,\boldsymbol{\lambda},\boldsymbol{\sigma})}$}
  \Step{\textbf{if} $\neg\OnTime(\mathrm{Grant})$ \textbf{then return}}
  \Step{$\Xi\gets\varnothing$}
  \Block{\textbf{for} $\jmath=0,\ldots,\varrho$ \textbf{do}}{%
    \Block{\textbf{for} $\iota\in\Theta_\jmath$ in slot order \textbf{do}}{%
      \Step{$\Delta_\iota\gets\Rx{\sigma_\iota}{\phi,\jmath,\iota}$}
      \Step{$\Xi\gets\Xi\cup\Deliver(\Delta_\iota)$}%
    }
    \Step{$\Pi_\jmath\gets\begin{cases}
      \Pick_{\varkappa}(\Missing(\Theta_0),\phi,\jmath), & \jmath<\varrho\\
      \varnothing, & \jmath=\varrho
    \end{cases}$}
    \Step{$\Tx{\ctrl}{\mathrm{ACK}(\phi,\jmath,\Xi,\Pi_\jmath)}$}
    \Step{$\Theta_{\jmath+1}\gets\begin{cases}
      \Pi_\jmath, & \OnTime(\mathrm{ACK})\\
      \varnothing, & \text{otherwise}
    \end{cases}$}%
  }
  \Step{$\Review_\iota(\phi)$ for each $\iota\in\Theta_0$ satisfying $\Eligible_\iota(\phi)$}
\end{algorithm}

\begin{algorithm}{2}{Client delivery}
  \Step{$(\phi,\Omega)\gets\Rx{\ctrl}{\mathrm{Beacon}}$}
  \Step{\textbf{if} $\neg\mathrm{Fresh}(\phi)$ \textbf{then return} \textsc{search}}
  \Step{\textbf{if} $\Delta_\iota=\bot$ \textbf{then return}}
  \Step{$\Tx{\ctrl}{\mathrm{Request}(\phi,\lambda_\iota,\phi_{\mathrm{ack}})}$}
  \Step{\textbf{if} Request TX did not start \textbf{then return} \textsc{req fail}}
  \Step{$\Gamma\gets\ValidGrant_\iota(\phi,\lambda_\iota)$}
  \Step{\textbf{if} $\Gamma=\bot$ \textbf{then return} \textsc{no grant}}
  \Step{$\sigma_\iota\gets\Gamma.\sigma_\iota,\quad \epsilon_0\gets\mathrm{true}$}
  \Block{\textbf{for} $\jmath=0,\ldots,\varrho$ \textbf{while} $\mathrm{Fresh}(\phi)$}{%
    \Block{\textbf{if} $\epsilon_\jmath\land(0\le\tau-\tau_{\iota,\jmath}<\delta_{\mathrm{tx}})$ \textbf{then}}{%
      \Step{$\Tx{\sigma_\iota}{\Delta_\iota}$}%
    }
    \Step{$(\Xi_\jmath,\Pi_\jmath)\gets\ValidACK_{\ctrl}(\phi,\jmath)$}
    \Block{\textbf{if} $\mathrm{Sent}(\Delta_\iota)\land\iota\in\Xi_\jmath$ \textbf{then}}{%
      \Step{$\phi_{\mathrm{ack}}\gets\phi$ \enspace\textbf{return} \textsc{accepted}}%
    }
    \Step{$\epsilon_{\jmath+1}\gets(\iota\in\Pi_\jmath)$}%
  }
  \Step{\textbf{if} $\neg\mathrm{Sent}(\Delta_\iota)$ \textbf{then return} \textsc{no tx}}
  \Step{\textbf{if} final feedback is valid \textbf{then return} \textsc{missing}}
  \Step{\textbf{return} \textsc{unknown}}
\end{algorithm}

\begin{algorithm}{3}{Gateway SF recovery and review}
  \Step{\textbf{At frame start:} $\Expire_\iota(\phi)$}
  \Block{\textbf{if} probing and $\phi-\phi_{\mathrm{probe}}>\nu_{\mathrm{ack}}$ \textbf{then}}{%
    \Step{$\Rollback(\sigma^{\mathrm{back}}_\iota,\nu_{\mathrm{hold}})$}
  }
  \Step{\textbf{Before the Grant:} $\Recover_\iota$}
  \Step{\textbf{if} session changed \textbf{then} reset state to $\sigma_{\min}$}
  \Block{\textbf{if} $\phi_{\mathrm{ack}}=\phi^{\mathrm{del}}_\iota\ne 0$ in this session \textbf{then}}{%
    \Step{$\sigma^{\mathrm{mem}}_\iota\gets
      \max(\sigma^{\mathrm{mem}}_\iota,\sigma^{\mathrm{del}}_\iota)$}%
  }
  \Block{\textbf{if} $\phi^{\mathrm{req}}_\iota\ne0$ and $\phi-\phi^{\mathrm{req}}_\iota\ge\nu_{\mathrm{idle}}$ \textbf{then}}{%
    \Step{$\sigma^{\mathrm{res}}_\iota\gets\sigma_{\min}$ and clear the sample history}%
  }
  \Step{\textbf{if} rejoining after expiry \textbf{then}
    $\sigma^{\mathrm{want}}_\iota\gets\sigma^{\mathrm{mem}}_\iota$}
  \vspace{3pt}
  \Step{\textbf{After feedback:} $\Review_\iota$ for an eligible report}
  \Step{On SF change, clear the loss history and first-delivery count}
  \Step{$r_\sigma\gets x_{\mathrm{req}}+b_\sigma-\vartheta_\sigma$}
  \Step{$e\gets\mathrm{retry\ rescue}\lor(\mathrm{no\ DATA}\land
      x_{\mathrm{req}}\ne\bot\land r_\sigma<M_{\mathrm{up}})$}
  \Block{\textbf{if} EWMA or Kalman \textbf{then}}{%
    \Step{Update $\widehat{x}$ and $\sigma_x$ on the Request SNR scale}
    \Step{$S\gets\rho S+w$, where $w=0$ if $\neg e$}%
  }
  \Block{\textbf{if} $\sigma_\iota<\sigma^{\mathrm{want}}_\iota$ \textbf{then}}{%
    \Step{\textbf{if} $\nu^{\mathrm{first}}_\iota\ge\nu_{\mathrm{lock}}\land\alpha_\iota$
      \textbf{then} $\Lock(\sigma_\iota)$}
    \Step{\textbf{return}}%
  }
  \Block{\textbf{if} probing and $(\epsilon_\iota^{\mathrm{fail}}
      \lor (\phi-\phi_{\mathrm{probe}}\ge\nu_{\mathrm{ack}}\land\neg\alpha_\iota))$ \textbf{then}}{%
    \Step{$\Rollback(\sigma^{\mathrm{back}}_\iota,\nu_{\mathrm{hold}})$}
    \Step{\textbf{return}}%
  }
  \Block{\textbf{if} Legacy \textbf{then}}{%
    \Step{$\boldsymbol{\xi}\gets\Tail_{\nu_{\mathrm{obs}}}(\boldsymbol{\xi},e)$}
    \Step{$u\gets\left(\sum_j\xi_j\ge\nu_{\mathrm{up}}\right)$}
    \Step{$d\gets(\overline{\upsilon}\ge\theta_{\sigma-1})$ for $\sigma>\sigma_{\min}$}
  }
  \Block{\textbf{else}}{%
    \Step{$m_\sigma\gets\widehat{x}+b_\sigma-\vartheta_\sigma-z\sigma_x$}
    \Step{$u\gets(S\ge S_{\mathrm{up}})\land(m_\sigma<M_{\mathrm{up}})$}
    \Step{$d\gets(m_{\sigma-1}\ge M_{\mathrm{down}})$ for $\sigma>\sigma_{\min}$}
  }
  \Block{\textbf{if} $\epsilon_\iota^{\mathrm{fail}}$ \textbf{then}}{%
    \Block{\textbf{if} $e\land u$ \textbf{then}}{%
      \Step{$\sigma^{\mathrm{want}}_\iota\gets\min(\sigma_\iota+1,\sigma_{\max})$}
      \Step{Unlock and clear the loss history}%
    }
    \Step{\textbf{return}}%
  }
  \Step{\textbf{if} unlocked and $\nu^{\mathrm{first}}_\iota\ge\nu_{\mathrm{lock}}\land\alpha_\iota$
    \textbf{then} $\Lock(\sigma_\iota)$}
  \Step{\textbf{if} $h_\iota>0$ \textbf{then} $h_\iota\gets h_\iota-1$ \enspace\textbf{return}}
  \Step{$\chi_\iota\gets\alpha_\iota\land(\sigma_\iota>\sigma_{\min})
    \land(\nu^{\mathrm{first}}_\iota\ge\nu_{\mathrm{probe}})$}
  \Step{\textbf{if} locked and $\chi_\iota\land d$ \textbf{then}
    $\Probe(\sigma_\iota-1)$ and $\phi_{\mathrm{probe}}\gets\phi$}
\end{algorithm}
